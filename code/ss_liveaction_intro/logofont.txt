  function drawLogo(word, cx, y, cell) {
    const cells = [];
    let cxp = 0;
    for (const ch of word) {
      if (ch === " ") { cxp += 3; continue; }
      LOGO_FONT[ch].forEach((row, r) => { for (let c = 0; c < 5; c++) if (row[c] === "#") cells.push([cxp + c, r]); });
